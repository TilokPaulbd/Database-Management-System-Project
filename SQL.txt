
CREATE TABLE Admins (
 admin_id VARCHAR(30) PRIMARY KEY,
 admin_name VARCHAR(30),
 designation VARCHAR(30),
 password VARCHAR(30)
);






CREATE TABLE Students (
    student_id VARCHAR(30) PRIMARY KEY,
    student_name VARCHAR(30),
    email VARCHAR(30),
    phone INT,
    password VARCHAR(30)
);









CREATE TABLE Semesters (
    semester_id INT PRIMARY KEY,
    semester_name VARCHAR(30)
);






CREATE TABLE Routes (
    route_id INT PRIMARY KEY,
    start_destination VARCHAR(30),
    end_destination VARCHAR(30)
);






CREATE TABLE Buses (
    bus_id INT PRIMARY KEY,
    bus_name VARCHAR(30)
);






CREATE TABLE Seats (
    seat_id INT AUTO_INCREMENT PRIMARY KEY,
    bus_id INT,
    seat_number INT,
    FOREIGN KEY (bus_id) REFERENCES Buses(bus_id)
);








CREATE TABLE Schedules (
    schedule_id INT PRIMARY KEY,
    route_id INT,
    journey_date DATE,
    start_time TIME,
    end_time TIME,
    FOREIGN KEY (route_id) REFERENCES Routes(route_id)
);









CREATE TABLE Bus_schedules ( 
    bus_schedule_id INT PRIMARY KEY, 
    bus_id INT, 
    schedule_id INT, 
    FOREIGN KEY (bus_id) REFERENCES Buses(bus_id), 
    FOREIGN KEY (schedule_id) REFERENCES Schedules(schedule_id) 
);











CREATE TABLE Bookings (
    booking_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id VARCHAR(30),
    seat_id INT,
    schedule_id INT,
    booking_time DATETIME,
    FOREIGN KEY (student_id) REFERENCES Students(student_id),
    FOREIGN KEY (seat_id) REFERENCES Seats(seat_id),
    FOREIGN KEY (schedule_id) REFERENCES Schedules(schedule_id)
);










CREATE TABLE Travel_status (
 travel_status_id INT AUTO_INCREMENT PRIMARY KEY,
 booking_id INT,
 travel_status VARCHAR(30) DEFAULT 'CONFIRMED',
 update_time DATETIME,
 FOREIGN KEY (booking_id) REFERENCES Bookings(booking_id)
);











CREATE TABLE Semesters (
    semester_id INT PRIMARY KEY ,
    semester_name VARCHAR(50)
);











CREATE TABLE Payments (
    payment_id INT PRIMARY KEY,
    student_id VARCHAR(30),
    semester_id INT,
    payment_status VARCHAR(30),
    FOREIGN KEY (student_id) REFERENCES Students(student_id),
    FOREIGN KEY (semester_id) REFERENCES Semesters(semester_id)
);











CREATE TABLE Travel_log (
 travel_log_id INT AUTO_INCREMENT PRIMARY KEY,
 booking_id INT,
 student_id VARCHAR(30),
 bus_schedule_id INT,
 seat_id INT,
 start_destination VARCHAR(30),
 end_destination VARCHAR(30),
 travel_status VARCHAR(30),
 log_time DATETIME
);









CREATE PROCEDURE get_student_travel_logs(IN p_student_id VARCHAR(30))
BEGIN


 SELECT 
 travel_log_id,
 booking_id,
 student_id,
 bus_schedule_id,
 seat_id,
 start_destination,
 end_destination,
 travel_status,
 log_time,

 FROM Travel_log
 WHERE student_id = p_student_id
 ORDER BY log_time DESC;

 

END $$

                                      procedure create_seats

DELIMITER $$

CREATE PROCEDURE create_seats(IN b_id INT)
BEGIN
    DECLARE i INT DEFAULT 1;

    WHILE i <= 40 DO
        INSERT INTO Seats(bus_id, seat_number)
        VALUES (b_id, i);
        SET i = i + 1;
    END WHILE;

END $$

DELIMITER ;

                                     trigger assign_bus_seats
DELIMITER $$

CREATE TRIGGER assign_bus_seats
AFTER INSERT ON Buses
FOR EACH ROW
BEGIN
    CALL create_seats(NEW.bus_id);
END $$

                                  procedure add_booking

DELIMITER $$

CREATE PROCEDURE add_booking(
    IN input_student_id VARCHAR(30),
    IN input_bus_schedule_id INT,
    IN input_seat_id INT
)
BEGIN
    DECLARE selected_schedule_id INT;

    SELECT schedule_id
    INTO selected_schedule_id
    FROM Bus_schedules
    WHERE bus_schedule_id = input_bus_schedule_id;

    IF EXISTS (
        SELECT *
        FROM available_seats
        WHERE seat_id = input_seat_id
    ) THEN

        INSERT INTO Bookings(
            student_id,
            seat_id,
            schedule_id,
            booking_time
        )
        VALUES (
            input_student_id,
            input_seat_id,
            selected_schedule_id,
            NOW()
        );

    END IF;

END $$

                                        PROCEDURE add_travel_status_with_log

                                        
DELIMITER $$

CREATE PROCEDURE add_travel_status_with_log(
    IN input_booking_id INT,
    IN input_status VARCHAR(30)
)
BEGIN
    DECLARE booking_student_id VARCHAR(30);
    DECLARE booking_schedule_id INT;
    DECLARE booking_seat_id INT;
    DECLARE bus_schedule_identifier INT;
    DECLARE route_start_destination VARCHAR(30);
    DECLARE route_end_destination VARCHAR(30);

   
    SELECT student_id, schedule_id, seat_id
    INTO booking_student_id, booking_schedule_id, booking_seat_id
    FROM Bookings
    WHERE booking_id = input_booking_id;


    SELECT bus_schedule_id
    INTO bus_schedule_identifier
    FROM Bus_schedules
    WHERE schedule_id = booking_schedule_id
    LIMIT 1;

 
    SELECT Routes.start_destination, Routes.end_destination
    INTO route_start_destination, route_end_destination
    FROM Schedules
    JOIN Routes 
        ON Schedules.route_id = Routes.route_id
    WHERE Schedules.schedule_id = booking_schedule_id;

   
    INSERT INTO Travel_status(
        booking_id,
        travel_status,
        update_time
    )
    VALUES (
        input_booking_id,
        input_status,
        NOW()
    );


    INSERT INTO Travel_log(
        booking_id,
        student_id,
        bus_schedule_id,
        seat_id,
        start_destination,
        end_destination,
        travel_status,
        log_time
    )
    VALUES (
        input_booking_id,
        booking_student_id,
        bus_schedule_identifier,
        booking_seat_id,
        route_start_destination,
        route_end_destination,
        input_status,
        NOW()
    );

END $$

                                        PROCEDURE update_travel_status






DELIMITER $$

CREATE PROCEDURE update_travel_status(
 IN input_booking_id INT,
 IN input_status VARCHAR(30)
)
BEGIN

 UPDATE Travel_status
 SET travel_status = input_status,
 update_time = NOW()
 WHERE booking_id = input_booking_id;

 UPDATE Travel_log
 SET travel_status = input_status,
 log_time = NOW()
 WHERE booking_id = input_booking_id;

 IF input_status = 'DENIED' THEN

 DELETE FROM Travel_status
 WHERE booking_id = input_booking_id;

 DELETE FROM Bookings
 WHERE booking_id = input_booking_id;

 END IF;

END $$

                                   PROCEDURE get_schedules_by_date





DELIMITER $$

CREATE PROCEDURE get_schedules_by_date(IN p_date DATE)
BEGIN
 SELECT 
 bs.bus_schedule_id,
 s.schedule_id,
 r.start_destination,
 r.end_destination,
 s.start_time,
 s.end_time,
 bs.bus_id
 FROM Bus_schedules bs
 JOIN Schedules s ON bs.schedule_id = s.schedule_id
 JOIN Routes r ON s.route_id = r.route_id
 WHERE s.journey_date = p_date;
END $$
